"use client";

import { useEffect } from "react";

/**
 * Sets intro phase on <html> only after mount â€” avoids SSR/client hydration mismatch.
 * Hero sets `dataset.heroIntro = "done"` when the hand-off finishes.
 * Adds intro-ready on body + html so globals.css can fade the veil and body without flash.
 */
export function IntroController() {
  useEffect(() => {
    document.documentElement.dataset.heroIntro = "running";
    document.documentElement.dataset.headerTheme = "hero";
    document.documentElement.classList.add("intro-ready");
    document.body.classList.add("intro-ready");

    return () => {
      document.documentElement.removeAttribute("data-hero-intro");
      document.documentElement.removeAttribute("data-header-theme");
    };
  }, []);

  return null;
}

"use client";

import { usePathname } from "next/navigation";
import { useEffect } from "react";

/**
 * Ensures intro veil / body fade CSS does not trap non-home routes with opacity 0.
 * Home relies on IntroController to add intro-ready after mount.
 */
export function IntroReadyGate() {
  const pathname = usePathname();

  useEffect(() => {
    if (pathname !== "/") {
      document.documentElement.classList.add("intro-ready");
      document.body.classList.add("intro-ready");
    }
  }, [pathname]);

  return null;
}

"use client";

import Image from "next/image";
import { createPortal } from "react-dom";
import { useEffect, useLayoutEffect, useRef, useState } from "react";
import { gsap } from "gsap";

const INTRO_HOLD_S = 1.05;
const INTRO_HANDOFF_S = 1.65;

/** Keep in sync with `.site-header { top }` in `globals.css`. */
const HEADER_ROW_TOP_PX = 24;
/**
 * Negative = shift wordmark up after handoff so raster caps align with `.nav-left` / CTA
 * (PNG padding + scale); tweak only here + in `finishIntro` / timeline together.
 */
const WORDMARK_REST_Y_NUDGE_PX = -26;

export default function HeroScrollShrink() {
  const sectionRef = useRef<HTMLElement>(null);
  const videoShellRef = useRef<HTMLDivElement>(null);
  const brandTitleRef = useRef<HTMLDivElement>(null);
  const videoRef = useRef<HTMLVideoElement>(null);
  const introBackdropRef = useRef<HTMLDivElement>(null);
  const [wordmarkInBody, setWordmarkInBody] = useState(false);

  useLayoutEffect(() => {
    setWordmarkInBody(true);
  }, []);

  useLayoutEffect(() => {
    const section = sectionRef.current;
    const brandTitle = brandTitleRef.current;
    const video = videoRef.current;
    const introBackdrop = introBackdropRef.current;

    if (!wordmarkInBody || !section || !brandTitle || !video || !introBackdrop)
      return;

    const prefersReducedMotion = window.matchMedia(
      "(prefers-reduced-motion: reduce)",
    ).matches;

    const ctx = gsap.context(() => {
      gsap.set(brandTitle, {
        position: "fixed",
        left: "50%",
        top: "50%",
        xPercent: -50,
        yPercent: -50,
        x: 0,
        y: 0,
        scale: 1,
        autoAlpha: 1,
        zIndex: 9999,
        transformOrigin: "50% 50%",
        force3D: true,
      });

      gsap.set(video, {
        opacity: prefersReducedMotion ? 1 : 0,
        scale: prefersReducedMotion ? 1 : 1.035,
        filter: "contrast(1.04) saturate(1.02) brightness(0.94)",
        force3D: true,
      });

      gsap.set(introBackdrop, {
        opacity: prefersReducedMotion ? 0 : 1,
      });

      const finishIntro = () => {
        document.documentElement.dataset.heroIntro = "done";

        gsap.set(brandTitle, {
          position: "fixed",
          left: "50%",
          top: HEADER_ROW_TOP_PX,
          xPercent: -50,
          yPercent: 0,
          x: 0,
          y: WORDMARK_REST_Y_NUDGE_PX,
          scale: 0.34,
          autoAlpha: 1,
          zIndex: 9999,
          transformOrigin: "50% 0%",
          force3D: true,
        });
      };

      if (prefersReducedMotion) {
        gsap.set(video, { opacity: 1, scale: 1 });
        gsap.set(introBackdrop, { opacity: 0 });
        finishIntro();
        return;
      }

      gsap
        .timeline({
          delay: INTRO_HOLD_S,
          onComplete: finishIntro,
        })
        .to(
          video,
          {
            opacity: 1,
            scale: 1,
            duration: INTRO_HANDOFF_S,
            ease: "power3.inOut",
          },
          0,
        )
        .to(
          introBackdrop,
          {
            opacity: 0,
            duration: INTRO_HANDOFF_S,
            ease: "power3.inOut",
          },
          0,
        )
        .to(
          brandTitle,
          {
            left: "50%",
            top: HEADER_ROW_TOP_PX,
            xPercent: -50,
            yPercent: 0,
            x: 0,
            y: WORDMARK_REST_Y_NUDGE_PX,
            scale: 0.34,
            autoAlpha: 1,
            zIndex: 9999,
            duration: INTRO_HANDOFF_S,
            ease: "power3.inOut",
            transformOrigin: "50% 0%",
            force3D: true,
          },
          0,
        );
    }, section);

    return () => {
      document.documentElement.removeAttribute("data-hero-intro");
      ctx.revert();
    };
  }, [wordmarkInBody]);

  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;

    video.muted = true;
    video.play().catch(() => {});
  }, []);

  useEffect(() => {
    const section = sectionRef.current;
    const shell = videoShellRef.current;
    if (!section || !shell) return;

    const prefersReducedMotion = window.matchMedia(
      "(prefers-reduced-motion: reduce)",
    ).matches;
    if (prefersReducedMotion) {
      document.documentElement.dataset.headerTheme = "hero";
      return () => {
        document.documentElement.removeAttribute("data-header-theme");
      };
    }

    let raf = 0;

    const clamp = (v: number, min: number, max: number) =>
      Math.min(Math.max(v, min), max);

    const update = () => {
      if (document.documentElement.dataset.heroIntro !== "done") {
        shell.style.transform = "";
        shell.style.borderRadius = "";
        section.style.setProperty("--hero-shrink-progress", "0");
        document.documentElement.dataset.headerTheme = "hero";
        return;
      }

      const rect = section.getBoundingClientRect();
      const vh = window.innerHeight || document.documentElement.clientHeight;
      const scrollable = Math.max(1, rect.height - vh);
      const progress = clamp(-rect.top / scrollable, 0, 1);

      const eased = 1 - (1 - progress) ** 3;
      section.style.setProperty("--hero-shrink-progress", String(eased));
      const scaleTop = 1;
      const scaleBottom = 0.28;
      const scale = scaleTop - eased * (scaleTop - scaleBottom);
      const radius = eased * 30;
      const y = eased * 38;

      shell.style.transform = `translate3d(0, ${y}px, 0) scale(${scale})`;
      shell.style.borderRadius = `${radius}px`;

      if (scale <= 0.82) {
        document.documentElement.dataset.headerTheme = "light";
      } else {
        document.documentElement.dataset.headerTheme = "hero";
      }
    };

    const onScroll = () => {
      cancelAnimationFrame(raf);
      raf = requestAnimationFrame(update);
    };

    update();
    window.addEventListener("scroll", onScroll, { passive: true });
    window.addEventListener("resize", onScroll);

    const introObs = new MutationObserver(() => {
      onScroll();
    });
    introObs.observe(document.documentElement, {
      attributes: true,
      attributeFilter: ["data-hero-intro"],
    });

    return () => {
      cancelAnimationFrame(raf);
      window.removeEventListener("scroll", onScroll);
      window.removeEventListener("resize", onScroll);
      introObs.disconnect();
      document.documentElement.removeAttribute("data-header-theme");
    };
  }, []);

  const wordmark = (
    <div
      ref={brandTitleRef}
      className="brand-title hero-wordmark-fixed hero-wordmark intro-wordmark intro-wordmark--hero intro-wordmark--raster select-none"
      role="img"
      aria-label="Four et DÃ©lices"
    >
      <span className="sr-only">Four et DÃ©lices</span>
      <Image
        src="/assets/four-et-delices-wordmark.png"
        alt=""
        width={3072}
        height={2046}
        sizes="(max-width: 768px) min(92vw, 560px), 560px"
        priority
        unoptimized
        className="intro-wordmark__img"
        aria-hidden
      />
    </div>
  );

  return (
    <>
      <section
        ref={sectionRef}
        aria-label="Introduction"
        className="hero-scroll-section w-full"
      >
        <div className="hero-sticky-stage">
          <div className="hero-scroll-editorial" aria-hidden>
            <span className="hero-scroll-editorial__line hero-scroll-editorial__line--1">Every</span>
            <span className="hero-scroll-editorial__line hero-scroll-editorial__line--2">Celebration</span>
            <span className="hero-scroll-editorial__line hero-scroll-editorial__line--3">Deserves</span>
            <span className="hero-scroll-editorial__line hero-scroll-editorial__line--4">
              a Masterpiece
            </span>
          </div>

          <div ref={videoShellRef} className="hero-video-shell">
            <video
              ref={videoRef}
              playsInline
              autoPlay
              muted
              loop
              preload="auto"
              controls={false}
              className="hero-video pointer-events-none opacity-0"
            >
              <source src="/assets/hero-video.mp4" type="video/mp4" />
              <source src="/assets/hero-video3.mp4" type="video/mp4" />
            </video>

            <div className="hero-video-overlay" aria-hidden />
          </div>

          <div
            ref={introBackdropRef}
            className="hero-intro-backdrop"
            aria-hidden
          />

}

const MIN_DISTANCE = 38;
const MAX_VISIBLE = 10;
const LIFE_MS = 1900;
const CARD_WIDTH = 360;
const CARD_HEIGHT = 440;

/** Align trail with custom knife cursor tip (blade vs mouse hotspot); see CustomCursor.tsx */
const CURSOR_TIP_OFFSET_X = 10;
const CURSOR_TIP_OFFSET_Y = 14;

const IMAGES = [
  "/assets/reveal/B-1-1.png",
  "/assets/reveal/B-2-1.png",

];

type TrailImage = {
  id: number;
  src: string;
  x: number;
  y: number;
  rotateDeg: number;
};

function shuffleImages(images: string[], previousLast?: string) {
  const shuffled = [...images];

  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }

  if (previousLast && shuffled[0] === previousLast && shuffled.length > 1) {
    [shuffled[0], shuffled[1]] = [shuffled[1], shuffled[0]];
  }

  return shuffled;
}

export default function TasteRevealSection() {
  const sectionRef = useRef<HTMLElement>(null);
  const trailRef = useRef<HTMLDivElement>(null);
  const idRef = useRef(0);
  const indexRef = useRef(0);
  const lastPointRef = useRef<{ x: number; y: number } | null>(null);
  const lastImageRef = useRef<string | undefined>(undefined);
  const sequenceRef = useRef<string[]>([]);
  const [trailImages, setTrailImages] = useState<TrailImage[]>([]);

  useEffect(() => {
    sequenceRef.current = shuffleImages(IMAGES);
  }, []);

  const staticImages = useMemo(() => IMAGES.slice(0, 5), []);

  const getNextImage = () => {
    if (!sequenceRef.current.length) {
      sequenceRef.current = shuffleImages(IMAGES, lastImageRef.current);
      indexRef.current = 0;
    }

    const src = sequenceRef.current[indexRef.current];
    indexRef.current += 1;

    if (indexRef.current >= sequenceRef.current.length) {
      lastImageRef.current = src;
      sequenceRef.current = shuffleImages(IMAGES, lastImageRef.current);
      indexRef.current = 0;
    } else {
      lastImageRef.current = src;
    }

    return src;
  };

  useEffect(() => {
    const section = sectionRef.current;
    const trail = trailRef.current;
    if (!section || !trail) return;

    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const finePointer = window.matchMedia("(pointer: fine) and (hover: hover)").matches;

    if (reducedMotion || !finePointer) return;

    let cancelled = false;

    const handlePointerMove = (event: PointerEvent) => {
      const rect = trail.getBoundingClientRect();
      const pointerY = event.clientY - rect.top;
      const redAreaStartY = 48;
      if (pointerY < redAreaStartY) return;

      const x = event.clientX - rect.left + CURSOR_TIP_OFFSET_X;
      const y = event.clientY - rect.top + CURSOR_TIP_OFFSET_Y;

      if (y < redAreaStartY) return;

      const lastPoint = lastPointRef.current;
      let rotateDeg = 0;
      if (lastPoint) {

        const dy = y - lastPoint.y;
        const distance = Math.sqrt(dx * dx + dy * dy);
        if (distance < MIN_DISTANCE) return;
        const pathAngleDeg = (Math.atan2(dy, dx) * 180) / Math.PI;
        rotateDeg = Math.max(-2, Math.min(2, pathAngleDeg * 0.028));
      }

      lastPointRef.current = { x, y };

      const src = getNextImage();

      const image: TrailImage = {
        id: idRef.current++,
        src,
        x,
        y,
        rotateDeg,
      };

      setTrailImages((prev) => [...prev, image].slice(-MAX_VISIBLE));

      window.setTimeout(() => {
        if (cancelled) return;
        setTrailImages((prev) => prev.filter((item) => item.id !== image.id));
      }, LIFE_MS);
    };

    section.addEventListener("pointermove", handlePointerMove);

    return () => {
      cancelled = true;
      section.removeEventListener("pointermove", handlePointerMove);
    };
  }, []);

  return (
    <section ref={sectionRef} className="taste-reveal" aria-label="Cake teaser">

            UNFORGETTABLE
          </span>
          <span className="taste-reveal__title-line taste-reveal__title-line--bottom">
            TASTE
          </span>
        </h2>
      </div>

      <div className="taste-reveal__red-zone" aria-hidden>

                ["--x" as string]: `${image.x}px`,
                ["--y" as string]: `${image.y}px`,

  "M506.8,14.319c-6.4-6.4-17.6-11.6-36-0.8c-91.6,55.2-225.6,180-227.2,181.2c-0.4,0.4-0.4,0.4-0.4,0.8l-0.4,0.4c0,0,0,0.4,0,0.8c0,0.4,0,0.4,0,0.8v44L9.2,452.719c-6,6-9.2,13.6-9.2,22c0,8.4,3.2,16,8.8,22c6.4,6,14.4,8.8,22,8.8c8,0,16-3.2,22-9.2l454-453.6C514.8,34.719,514.8,21.919,506.8,14.319z";

/** Lower = smoother/luxury lag (lerp toward mouse); higher = snappier */
const SMOOTH = 0.11;

const CLICK_MS = 150;

/** Grow cursor when hovering interactive targets (DOM â€” not CSS ~ sibling) */
const INTERACTIVE_FOR_HOVER =
  'button, a[href], [role="button"], input[type="submit"], input[type="button"], input[type="reset"], summary';

export function CustomCursor() {
  const rootRef = useRef<HTMLDivElement>(null);
  const innerRef = useRef<HTMLDivElement>(null);
  const svgRef = useRef<SVGSVGElement>(null);
  /** Blade tip offset from root TL (px), includes inner CSS transforms */
  const hotspotRef = useRef({ hx: TIP_X, hy: TIP_Y });
  const clickAnimRef = useRef(0);
  const posRef = useRef({ x: 0, y: 0 });
  const visualRef = useRef({ x: 0, y: 0 });
  const syncedRef = useRef(false);
  const rafRef = useRef(0);
  const [active, setActive] = useState(false);

  const measureHotspot = useCallback(() => {
    const root = rootRef.current;
    const svg = svgRef.current;
    if (!root || !svg) return;
    const pt = svg.createSVGPoint();
    pt.x = TIP_UX;
    pt.y = TIP_UY;
    const m = svg.getScreenCTM();
    if (!m) return;
    const scr = pt.matrixTransform(m);
    const rr = root.getBoundingClientRect();
    hotspotRef.current = {
      hx: scr.x - rr.left,
      hy: scr.y - rr.top,
    };
  }, []);

  const flushTransform = useCallback(() => {
    const el = rootRef.current;
    if (!el) return;
    const v = visualRef.current;
    const { hx, hy } = hotspotRef.current;
    el.style.transform = `translate3d(${v.x - hx}px, ${v.y - hy}px, 0)`;
  }, []);

  /** Tip vs root offset depends on globals.css rotate/scale â€” measure after paint */
  useLayoutEffect(() => {
    if (!active) return;
    const schedule = () => {
      requestAnimationFrame(() => {
        requestAnimationFrame(() => {
          measureHotspot();
          flushTransform();
        });
      });
    };
    schedule();
    window.addEventListener("resize", schedule);
    return () => window.removeEventListener("resize", schedule);
  }, [active, measureHotspot, flushTransform]);

  useEffect(() => {
    const mq = window.matchMedia(CURSOR_MQ);
    const syncActive = () => setActive(mq.matches);

    syncActive();
    mq.addEventListener("change", syncActive);

    return () => mq.removeEventListener("change", syncActive);
  }, []);

  useEffect(() => {
    if (!active) return;

    const html = document.documentElement;
    html.setAttribute("data-custom-cursor", "true");

    const onMove = (e: MouseEvent) => {
      posRef.current = { x: e.clientX, y: e.clientY };
      const inner = innerRef.current;
      if (inner) {
        const hit = document.elementFromPoint(e.clientX, e.clientY);
        const grow =
          !!hit?.closest(INTERACTIVE_FOR_HOVER) &&
          !hit?.closest("[data-no-cursor-grow]");
        inner.classList.toggle("custom-knife-cursor-inner--hover", grow);
      }
      if (!syncedRef.current) {
        visualRef.current = { x: e.clientX, y: e.clientY };
        syncedRef.current = true;
        flushTransform();
      }
    };

    /** Left button only â€” no animation on right/middle; never preventDefault (forms/links untouched). */
    const onMouseDown = (e: MouseEvent) => {
      if (e.button !== 0) return;
      const inner = innerRef.current;
      if (!inner) return;
      inner.classList.add("custom-knife-cursor-inner--click");
      window.clearTimeout(clickAnimRef.current);
      clickAnimRef.current = window.setTimeout(() => {
        inner.classList.remove("custom-knife-cursor-inner--click");
      }, CLICK_MS);
    };

    const loop = () => {
      const el = rootRef.current;
      if (el && syncedRef.current) {
        const t = posRef.current;
        const v = visualRef.current;
        v.x += (t.x - v.x) * SMOOTH;
        v.y += (t.y - v.y) * SMOOTH;
        const { hx, hy } = hotspotRef.current;
        el.style.transform = `translate3d(${v.x - hx}px, ${v.y - hy}px, 0)`;
      }
      rafRef.current = requestAnimationFrame(loop);
    };

    document.addEventListener("mousemove", onMove, { passive: true });
    document.addEventListener("mousedown", onMouseDown);
    rafRef.current = requestAnimationFrame(loop);

    return () => {
      html.removeAttribute("data-custom-cursor");
      document.removeEventListener("mousemove", onMove);
      document.removeEventListener("mousedown", onMouseDown);
      window.clearTimeout(clickAnimRef.current);
      cancelAnimationFrame(rafRef.current);
      syncedRef.current = false;
    };
  }, [active, flushTransform]);

  /** Same-origin Sweet Catch iframe: pointer events hit the iframe document, not ours â€” relay via postMessage. */
  useEffect(() => {
    if (!active) return;

    const onMsg = (e: MessageEvent) => {
      if (e.origin !== window.location.origin) return;
      const d = e.data;
      if (!d || typeof d !== "object") return;

      if (d.type === "sweet-catch-pointer") {
        const vx = typeof d.vx === "number" ? d.vx : 0;
        const vy = typeof d.vy === "number" ? d.vy : 0;
        posRef.current = { x: vx, y: vy };
        const inner = innerRef.current;
        if (inner != null && typeof d.grow === "boolean") {
          inner.classList.toggle("custom-knife-cursor-inner--hover", d.grow);
        }
      } else if (d.type === "sweet-catch-pointer-leave") {
        innerRef.current?.classList.remove("custom-knife-cursor-inner--hover");
      } else if (d.type === "sweet-catch-click") {
        const inner = innerRef.current;
        if (inner == null) return;
        inner.classList.add("custom-knife-cursor-inner--click");
        window.clearTimeout(clickAnimRef.current);
        clickAnimRef.current = window.setTimeout(() => {
          inner.classList.remove("custom-knife-cursor-inner--click");
        }, CLICK_MS);
      }
    };

    window.addEventListener("message", onMsg);
    return () => window.removeEventListener("message", onMsg);
  }, [active]);

  if (!active) return null;

  return (
    <div
      ref={rootRef}
      className="custom-cursor pointer-events-none fixed left-0 top-0 z-[9999] overflow-visible will-change-transform"
      style={{ transform: "translate3d(0px, 0px, 0)" }}
      aria-hidden
    >
      <div ref={innerRef} className="custom-knife-cursor-inner">
        <svg
          ref={svgRef}
          width={KNIFE_PX}
          height={KNIFE_PX}
          viewBox={`0 0 ${VIEWBOX} ${VIEWBOX}`}
          xmlns="http://www.w3.org/2000/svg"
          className="custom-knife-cursor-svg block overflow-visible"
          fill="currentColor"
        >
          <path d={KNIFE_PATH_D} />
        </svg>
      </div>
    </div>
  );
}
