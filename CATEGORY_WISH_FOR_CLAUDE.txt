export function CustomCursor() {
  const rootRef = useRef<HTMLDivElement>(null);
  const innerRef = useRef<HTMLDivElement>(null);
  const svgRef = useRef<SVGSVGElement>(null);
  const genieImgRef = useRef<HTMLImageElement>(null);
  /** Blade tip / genie face offset from root TL (px), includes inner CSS transforms */
  const hotspotRef = useRef({ hx: TIP_X, hy: TIP_Y });
  const clickAnimRef = useRef(0);
  const posRef = useRef({ x: 0, y: 0 });
  const visualRef = useRef({ x: 0, y: 0 });
  const syncedRef = useRef(false);
  const rafRef = useRef(0);
  const [active, setActive] = useState(false);
  const [cursorUi, setCursorUi] = useState<{
    mode: "knife" | "genie";
    mood: CategoryWishMood | null;
    grow: boolean;
    knifeClick: boolean;
    genieWink: boolean;
  }>({
    mode: "knife",
    mood: null,
    grow: false,
    knifeClick: false,
    genieWink: false,
  });

  const measureKnifeHotspot = useCallback(() => {
    const root = rootRef.current;
    const svg = svgRef.current;
    if (!root || !svg) return;
    const pt = svg.createSVGPoint();
    pt.x = TIP_UX;
    pt.y = TIP_UY;
    const m = svg.getScreenCTM();
    if (!m) return;
    const scr = pt.matrixTransform(m);
    const rr = root.getBoundingClientRect();
    hotspotRef.current = {
      hx: scr.x - rr.left,
      hy: scr.y - rr.top,
    };
  }, []);

  /** Face center (between â€œeyesâ€) from measured bitmap box */
  const measureGenieHotspot = useCallback(() => {
    const root = rootRef.current;
    const img = genieImgRef.current;
    if (!root || !img) return;

  );
